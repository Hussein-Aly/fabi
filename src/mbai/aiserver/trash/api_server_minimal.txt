@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    """catches all exceptions before response"""
    custom_exc_msg: str = CustomException.from_exception(exc).get_msg()
    logger.error(custom_exc_msg)
    return JSONResponse(
        status_code=500,
        content={"detail": "Internal Server Error", "message": custom_exc_msg},
    )

base_ai_router = APIRouter(prefix="/base", tags=["Base AI Routes"], dependencies=[])


@base_ai_router.post("/vector-search")
def route_vector_search(
    mb_connection_params: MBConnectionAPIParams,
    vs_params: VectorSearchParams,
    response_options: ResponseOptions = ResponseOptions(),
):
    mb_connection_values = build_mb_connection_values(mb_connection_params)
    bc = BaseAIController(mb_connection_values)

    ctlr_response = bc.vector_search(
        queries=vs_params.queries,
        k=vs_params.k,
        minscore=vs_params.minscore,
        resolve_overlaps=vs_params.resolve_overlaps,
        constraint=vs_params.constraint,
        additional_properties=vs_params.additional_properties,
        raise_on_error=True,
    )

    if response_options.verbosity == "minimal":
        ctlr_response = ctlr_response.to_minimal()
        ctlr_response.output = reduce_verbosity_snippet_list(ctlr_response.output)

    return ctlr_response


@base_ai_router.post("/llm_call_via_post")
def route_llm_call_via_post(
    llm_call_params: LLMCallParams,
    mb_connection_params: Optional[MBConnectionAPIParams] = None,
    response_options: ResponseOptions = ResponseOptions(),
):
    """post request for LLM directly, you can provide a inspire frontend url in llm_all_params.llm_url,
    but then you also need the mb_connection_params"""
    if llm_call_params.is_frontend_url() and not mb_connection_params:
        raise ValueError("if llm_url is a inspire frontend_url, you need to add mb_connection_params")

    if mb_connection_params:
        mb_connection_values = build_mb_connection_values(mb_connection_params)
    else:
        mb_connection_values = MBConnectionValues()

    bc = BaseAIController(mb_connection_values)

    ctlr_response = bc.llm_call_via_post_request(
        user_prompt=llm_call_params.user_prompt,
        model_json_schema=llm_call_params.response_json_schema,
        images=llm_call_params.images,
        llm_url=llm_call_params.llm_url,
        system_prompt=llm_call_params.system_prompt,
        model=llm_call_params.model,
        raise_on_error=True ** llm_call_params.get_additional_params(),
    )

    if response_options.verbosity == "minimal":
        ctlr_response = ctlr_response.to_minimal()

    return ctlr_response

class ExamplePersonData(BaseModel):
    age: int
    city: str


@example_router.get("/get_person_infos/{name}")
def get_person_infos(name: str = Path(..., description="this is the name (e.g. 'john' or anne)")) -> ExamplePersonData:
    """Get example data"""

    # .. do something with 'key'

    data = dict(john=ExamplePersonData(age=20, city="Vienna"), anne=ExamplePersonData(age=21, city="Linz"))
    return data[name.lower()]


@example_router.post("/set_infos/{name}")
def add_person_info(
    name: str = Path(..., description="this is the name (not 'john' or anne)"),
    person_data: ExamplePersonData = Body(..., description="this is the person data"),  # auto Validation for this class
) -> dict:
    """temporarily set example person data"""

    data = dict(john=ExamplePersonData(age=20, city="Vienna"), anne=ExamplePersonData(age=21, city="Linz"))

    if name.lower() not in data:
        data[name.lower()] = person_data

    return data
