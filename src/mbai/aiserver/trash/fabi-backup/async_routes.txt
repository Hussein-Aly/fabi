task_store = dict()
import time


class AsyncLogEntry(BaseModel):
    message: str
    ts: int = time.time()


class AsyncData(BaseModel):
    task_id: str
    start_time: int = time.time()
    status: Literal["pending", "running", "done"] = "pending"
    log: list = []
    result: str = "final result ????"


@app.get("/tasks/{task_id}", response_model=AsyncData)
async def get_task(task_id: str):
    """Get task status and result"""
    if task_id not in task_store:
        raise HTTPException(status_code=404, detail="Task not found")

    return task_store[task_id]


async def process_agent(task_id: str):
    """Simulate async task processing"""
    task_store[task_id].status = "running"

    for x in range(10):
        task_store[task_id].log.append(AsyncLogEntry(message=f"doing {x+1}/10"))
        await asyncio.sleep(2)

    # Update task with result
    task_store[task_id]["status"] = "done"
    task_store[task_id]["result"] = f"Task '{task_id}' completed successfully!"


@app.post("/agent", summary="")
async def route_agent(user_input: str) -> dict:
    task_id = str(uuid.uuid4())
    task_store[task_id] = AsyncData(
        task_id=task_id,
        log=[AsyncLogEntry(message=f"user-input is {user_input}")],
    )

    asyncio.create_task(process_agent(task_id))

    return dict(task_id=task_id)
